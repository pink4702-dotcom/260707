    <script>
        let timerInterval;
        let totalSeconds = 0;
        let isPaused = false;
        let playedWarning = false; // 10초 예비알림 중복 방지용

        const minutesInput = document.getElementById('minutes');
        const secondsInput = document.getElementById('seconds');
        
        const display = document.getElementById('timerDisplay');
        const startBtn = document.getElementById('startBtn');
        const pauseBtn = document.getElementById('pauseBtn');
        const resetBtn = document.getElementById('resetBtn');
        const inputSection = document.getElementById('inputSection');

        const warningSound = document.getElementById('warningSound');
        const alarmSound = document.getElementById('alarmSound');

        function updateDisplay(mins, secs) {
            const m = String(mins).padStart(2, '0');
            const s = String(secs).padStart(2, '0');
            display.textContent = `${m}:${s}`;
        }

        function startTimer() {
            if (!isPaused) {
                const mins = parseInt(minutesInput.value) || 0;
                const secs = parseInt(secondsInput.value) || 0;

                totalSeconds = (mins * 60) + secs;

                if (totalSeconds <= 0) {
                    alert("시간을 입력해 주세요! 🐻");
                    return;
                }
                playedWarning = false;
            }

            inputSection.style.visibility = 'hidden'; 
            startBtn.style.display = 'none';
            pauseBtn.style.display = 'inline-block';
            isPaused = false;

            timerInterval = setInterval(() => {
                if (totalSeconds <= 0) {
                    clearInterval(timerInterval);
                    updateDisplay(0, 0);
                    display.classList.remove('urgent');
                    
                    // [기능 2] 시간이 다 끝나면 경보음 재생
                    alarmSound.play().catch(e => console.log("오디오 재생 재생 오류:", e));
                    
                    setTimeout(() => {
                        alert("🎉 시간이 다 됐어요! 야호! 🎉");
                        resetTimer();
                    }, 100);
                    return;
                }

                totalSeconds--;

                const m = Math.floor(totalSeconds / 60);
                const s = totalSeconds % 60;

                updateDisplay(m, s);

                // [기능 1 & 3] 10초 이하로 남았을 때 조건 처리
                if (totalSeconds <= 10) {
                    display.classList.add('urgent'); // [기능 3] 빨간색으로 깜빡임
                    
                    if (totalSeconds === 10 && !playedWarning) {
                        // [기능 1] 정확히 10초가 되었을 때 예비알림(띠링) 재생
                        warningSound.play().catch(e => console.log("오디오 재생 오류:", e));
                        playedWarning = true;
                    }
                } else {
                    display.classList.remove('urgent');
                }
            }, 1000);
        }

        function pauseTimer() {
            clearInterval(timerInterval);
            isPaused = true;
            startBtn.textContent = '다시 출발! ▶️';
            startBtn.style.display = 'inline-block';
            pauseBtn.style.display = 'none';
            display.classList.remove('urgent');
        }

        function resetTimer() {
            clearInterval(timerInterval);
            totalSeconds = 0;
            isPaused = false;
            playedWarning = false;
            
            updateDisplay(0, 0);
            display.classList.remove('urgent');
            
            inputSection.style.visibility = 'visible';
            startBtn.textContent = '출발~! 🎈';
            startBtn.style.display = 'inline-block';
            pauseBtn.style.display = 'none';
            
            minutesInput.value = 0;
            secondsInput.value = 0;

            // 재생 중인 소리 멈춤
            alarmSound.pause();
            alarmSound.currentTime = 0;
            warningSound.pause();
            warningSound.currentTime = 0;
        }

        startBtn.addEventListener('click', startTimer);
        pauseBtn.addEventListener('click', pauseTimer);
        resetBtn.addEventListener('click', resetTimer);
    </script>
